Surveys & Feedback — Design

Pulse surveys, engagement scoring, ties into Performance reviews

Scope

  • Manager creates surveys with multiple-choice + text questions
  • Surveys assigned to specific employees or departments
  • Anonymous or named responses
  • Survey results page with aggregate scores
  • Employee sees assigned surveys on dashboard + dedicated page

Data Model

model Survey {
  id          String   @id @default(cuid())
  title       String
  description String?
  isAnonymous Boolean  @default(false)
  isActive    Boolean  @default(true)
  dueDate     DateTime?
  createdById String
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  createdBy   User     @relation(...)
  questions   SurveyQuestion[]
  assignments SurveyAssignment[]
}

model SurveyQuestion {
  id        String  @id @default(cuid())
  surveyId  String
  type      String  // "RATING" | "MULTIPLE_CHOICE" | "TEXT"
  question  String
  options   Json?   // for MULTIPLE_CHOICE
  order     Int
  survey    Survey  @relation(...)
  responses SurveyResponse[]
}

model SurveyAssignment {
  id         String   @id @default(cuid())
  surveyId   String
  employeeId String
  status     String   @default("PENDING") // PENDING | COMPLETED
  completedAt DateTime?
  survey     Survey   @relation(...)
  employee   Employee @relation(...)
}

model SurveyResponse {
  id         String  @id @default(cuid())
  questionId String
  assignmentId String?
  value      Json    // depends on type: number for RATING, string for TEXT, string[] for MULTIPLE_CHOICE
  question   SurveyQuestion @relation(...)
}
